function avaliarTier(tier, estado) {
  const feitos   = tier.ok + tier.falhas;
  const taxaErro = tier.falhas / Math.max(feitos, 1);

  if (feitos >= 5 && taxaErro > LIMITE_ERRO)
    return 'COMMIT_ISSUE';

  if (tier.ok / tier.total >= LIMIAR_PROGRESSAO)
    return 'LIBERAR_PROXIMO';

  if (feitos === tier.total)
    return 'COMMIT_ISSUE';

  return 'ESPERAR';
}

function gatePodeAvancar(servico, estado) {
  return estado.status !== 'COMMIT_ISSUE'
      && servico.tier >= estado.tierLiberado
      && tempoMinimoCumprido(servico)
      && dentroDaJanelaDeDeploy(servico)
      && !alertaDisparando(servico);
}
